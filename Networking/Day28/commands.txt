# Day 28 — Commands Used #

# Check Python backend servers
ps aux | grep "python3 -m http.server"

# Check Nginx service
sudo systemctl status nginx

# Open Nginx load balancer configuration
sudo nano /etc/nginx/sites-available/loadbalancer

# Test Nginx configuration
sudo nginx -t

# Enable load balancer configuration
sudo ln -s /etc/nginx/sites-available/loadbalancer /etc/nginx/sites-enabled/loadbalancer

# Reload Nginx after configuration changes
sudo systemctl reload nginx

# Check whether Nginx is listening on port 8080
sudo ss -tulnp | grep 8080

# Test Nginx default server
curl http://localhost

# Test the Nginx load balancer
curl http://localhost:8080

# Send multiple requests to the load balancer
for i in {1..6}; do curl -s http://localhost:8080; echo; done
